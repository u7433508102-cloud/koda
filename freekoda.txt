local function toggleFB()
    if cfg.fullbrightEnabled then
        Lighting.Brightness    = 2
        Lighting.ClockTime     = 14
        Lighting.GlobalShadows = false
        Lighting.Ambient       = Color3.fromRGB(178,178,178)
        Lighting.OutdoorAmbient= Color3.fromRGB(178,178,178)
    else
        Lighting.Brightness    = origL.Brightness
        Lighting.ClockTime     = origL.ClockTime
        Lighting.GlobalShadows = origL.GlobalShadows
        Lighting.Ambient       = origL.Ambient
        Lighting.OutdoorAmbient= origL.OutdoorAmbient
    end
end
 
local function setLightingMode(mode)
    if lightingConn then lightingConn:Disconnect(); lightingConn = nil end
    cfg.lightingMode = mode
    if mode == "Day" then
        Lighting.ClockTime = 14; Lighting.Brightness = 2
    elseif mode == "Night" then
        Lighting.ClockTime = 0;  Lighting.Brightness = 0
        Lighting.Ambient = Color3.fromRGB(30,30,30)
    elseif mode == "Sunset" then
        Lighting.ClockTime = 18; Lighting.Brightness = 1
        Lighting.Ambient = Color3.fromRGB(255,120,60)
    end
end
 
local function setFog(v)
    if v then
        Lighting.FogStart = cfg.fogStart
        Lighting.FogEnd   = cfg.fogEnd
        Lighting.FogColor = cfg.fogColor
    else
        Lighting.FogStart = origL.FogStart
        Lighting.FogEnd   = origL.FogEnd
        Lighting.FogColor = origL.FogColor
    end
end
 
local function fpsBoost()
    for _, obj in ipairs(game:GetService("Workspace"):GetDescendants()) do
        if obj:IsA("BasePart") then obj.CastShadow = false end
    end
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e8
end

local currentSky = nil
local function applySky(name)
    if skyboxConn then skyboxConn:Disconnect(); skyboxConn = nil end
    if currentSky then currentSky:Destroy(); currentSky = nil end
    if name == nil then return end
    local sky = Instance.new("Sky", Lighting)
    currentSky = sky
    local data = skies[name]
    if data then for k, v in pairs(data) do sky[k] = v end end
end
 
local function toggleRotatingSkybox(v)
    cfg.rotatingSkybox = v
    if skyboxConn then skyboxConn:Disconnect(); skyboxConn = nil end
    if v and currentSky then
        local RunService = game:GetService("RunService")
        local angle = 0
        skyboxConn = RunService.Heartbeat:Connect(function(dt)
            angle = angle + dt * 10
            Lighting.GeographicLatitude = angle % 360
        end)
    end
end

local accentObjects = {}
local function trackAccent(inst, prop)
    table.insert(accentObjects, {inst=inst, prop=prop})
end
local function applyAccent(color)
    cfg.accentColor = color
    theme.accent = color
    local h,s,v = Color3.toHSV(color)
    theme.accentDim = Color3.fromHSV(h, s*0.6, v*0.55)
    for _, entry in ipairs(accentObjects) do
        if entry.inst and entry.inst.Parent then
            entry.inst[entry.prop] = color
        end
    end
end

local function deselectMenuBtn()
    menuBtnSelected = false
    if menuBtn then
        menuBtn.BackgroundColor3 = Color3.fromRGB(0,0,0)
        menuBtn.BackgroundTransparency = 1
        for _, child in ipairs(menuBtn:GetChildren()) do
            if child:IsA("TextLabel") then
                if child.Name == "Icon" then child.TextColor3 = theme.muted
                elseif child.Name == "Lbl" then child.TextColor3 = theme.muted end
            end
        end
        local ind = menuBtn:FindFirstChild("Indicator")
        if ind then ind.Visible = false end
    end
    if menuPage then menuPage.Visible = false end
end

local function selectMenuBtn()
    menuBtnSelected = true
    for i, pg in ipairs(tabPages) do pg.Visible = false end
    for i, btn in ipairs(tabBtns) do
        btn.BackgroundColor3 = Color3.fromRGB(0,0,0)
        btn.BackgroundTransparency = 1
        for _, child in ipairs(btn:GetChildren()) do
            if child:IsA("TextLabel") then
                if child.Name == "Icon" then child.TextColor3 = theme.muted
                elseif child.Name == "Lbl" then child.TextColor3 = theme.muted end
            end
        end
        local ind = btn:FindFirstChild("Indicator")
        if ind then ind.Visible = false end
    end
    if menuBtn then
        menuBtn.BackgroundColor3 = theme.tabSel
        menuBtn.BackgroundTransparency = 0
        for _, child in ipairs(menuBtn:GetChildren()) do
            if child:IsA("TextLabel") then
                if child.Name == "Icon" then child.TextColor3 = theme.accent
                elseif child.Name == "Lbl" then child.TextColor3 = theme.text end
            end
        end
        local ind = menuBtn:FindFirstChild("Indicator")
        if ind then ind.Visible = true end
    end
    if menuPage then menuPage.Visible = true end
end
 
local function switchTab(idx)
    closeActiveDropdown()
    deselectMenuBtn()
    for i, pg in ipairs(tabPages) do pg.Visible = (i == idx) end
    for i, btn in ipairs(tabBtns) do
        local isSelected = (i == idx)
        btn.BackgroundColor3 = isSelected and theme.tabSel or Color3.fromRGB(0,0,0)
        btn.BackgroundTransparency = isSelected and 0 or 1
        for _, child in ipairs(btn:GetChildren()) do
            if child:IsA("TextLabel") then
                if child.Name == "Icon" then
                    child.TextColor3 = isSelected and theme.accent or theme.muted
                elseif child.Name == "Lbl" then
                    child.TextColor3 = isSelected and theme.text or theme.muted
                end
            end
        end
        local ind = btn:FindFirstChild("Indicator")
        if ind then ind.Visible = isSelected end
    end
end

    lbl.BackgroundTransparency = 1
    lbl.Text = label
    lbl.Font = Enum.Font.Code
    lbl.TextSize = 11

    lbl.TextXAlignment = Enum.TextXAlignment.Left

    valLbl.Font = Enum.Font.Code
    valLbl.TextSize = 11
    valLbl.TextColor3 = theme.muted
    valLbl.TextXAlignment = Enum.TextXAlignment.Right
    local trackBg = Instance.new("Frame", f)
    trackBg.Size = UDim2.new(1, -20, 0, 4)
    trackBg.Position = UDim2.new(0, 10, 0, 30)
    trackBg.BackgroundColor3 = theme.border
    trackBg.BorderSizePixel = 0
    Instance.new("UICorner", trackBg).CornerRadius = UDim.new(0, 2)
    local fill = Instance.new("Frame", trackBg)
    fill.BackgroundColor3 = theme.accent
    fill.BorderSizePixel = 0
    fill.Size = UDim2.new(math.clamp((cfg[cfgKey]-min)/(max-min),0,1), 0, 1, 0)
    Instance.new("UICorner", fill).CornerRadius = UDim.new(0, 2)
    trackAccent(fill, "BackgroundColor3")
    local thumb = Instance.new("Frame", trackBg)
    thumb.Size = UDim2.new(0, 10, 0, 10)
    local tv = math.clamp((cfg[cfgKey]-min)/(max-min),0,1)
    thumb.Position = UDim2.new(tv, -5, 0.5, -5)
    thumb.BackgroundColor3 = Color3.fromRGB(255,255,255)
    thumb.BorderSizePixel = 0
    Instance.new("UICorner", thumb).CornerRadius = UDim.new(0, 5)
    local dragging = false
    local function update(x)
        local t = math.clamp((x - trackBg.AbsolutePosition.X) / trackBg.AbsoluteSize.X, 0, 1)
        local v = math.floor((min + t*(max-min))/step + 0.5)*step
        v = math.clamp(math.floor(v*1000+0.5)/1000, min, max)
        cfg[cfgKey] = v
        local frac = (v-min)/(max-min)
        fill.Size = UDim2.new(frac, 0, 1, 0)
        thumb.Position = UDim2.new(frac, -5, 0.5, -5)

    end

        if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging=true; update(i.Position.X) end
    end)
    trackBg.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 then dragging=false end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and i.UserInputType == Enum.UserInputType.MouseMovement then update(i.Position.X) end
    end)
end

-- ── WALKSPEED LOGIC ───────────────────────────────────────────────────
local function applyWalkspeed()
    local char = lp.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    hum.WalkSpeed = cfg.walkspeedValue
    if walkspeedConn then walkspeedConn:Disconnect() end
    walkspeedConn = hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
        if cfg.walkspeedException then
            hum.WalkSpeed = cfg.walkspeedValue
        end
    end)
end
 
local function disableWalkspeed()
    local char = lp.Character
    if walkspeedConn then walkspeedConn:Disconnect(); walkspeedConn = nil end
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum.WalkSpeed = 16 end
end
 
lp.CharacterAdded:Connect(function()
    if cfg.walkspeedException then
        task.wait(0.5)
        applyWalkspeed()
    end
end)

-- ── ROBUST JSON ENCODER ───────────────────────────────────────────────
local function jsonEncode(val)
    local t = type(val)
    if val == nil then
        return "null"
    elseif t == "boolean" then
        return val and "true" or "false"
    elseif t == "number" then
        if val ~= val then return "null" end  -- NaN guard
        return string.format("%.6g", val)
    elseif t == "string" then
        -- escape special characters
        local escaped = val
            :gsub('\\', '\\\\')
            :gsub('"',  '\\"')
            :gsub('\n', '\\n')
            :gsub('\r', '\\r')
            :gsub('\t', '\\t')
        return '"' .. escaped .. '"'
    elseif t == "table" then
        local parts = {}
        -- check if array-like
        local isArray = true
        local maxN = 0
        for k, _ in pairs(val) do
            if type(k) ~= "number" or k ~= math.floor(k) or k < 1 then
                isArray = false; break
            end
            if k > maxN then maxN = k end
        end
        if isArray and maxN == #val then
            for i = 1, #val do
                parts[i] = jsonEncode(val[i])
            end
            return "[" .. table.concat(parts, ",") .. "]"
        else
            local i = 0
            for k, v in pairs(val) do
                i += 1
                parts[i] = jsonEncode(tostring(k)) .. ":" .. jsonEncode(v)
            end
            return "{" .. table.concat(parts, ",") .. "}"
        end
    end
    return "null"
end

-- ── ROBUST JSON DECODER ───────────────────────────────────────────────
local function jsonDecode(s)
    local pos = 1

    local function skipWS()
        while pos <= #s and s:sub(pos,pos):match("%s") do pos += 1 end
    end

    local parseValue  -- forward declare

    local function parseString()
        pos += 1  -- skip opening "
        local result = {}
        while pos <= #s do
            local c = s:sub(pos, pos)
            if c == '"' then pos += 1; break
            elseif c == '\\' then
                pos += 1
                local esc = s:sub(pos, pos)
                if     esc == '"'  then result[#result+1] = '"'
                elseif esc == '\\' then result[#result+1] = '\\'
                elseif esc == '/'  then result[#result+1] = '/'
                elseif esc == 'n'  then result[#result+1] = '\n'
                elseif esc == 'r'  then result[#result+1] = '\r'
                elseif esc == 't'  then result[#result+1] = '\t'
                else result[#result+1] = esc end
                pos += 1
            else
                result[#result+1] = c
                pos += 1
            end
        end
        return table.concat(result)
    end

    local function parseNumber()
        local numStr = s:match("^-?%d+%.?%d*[eE]?[+-]?%d*", pos)
        pos += #numStr
        return tonumber(numStr)
    end

    local function parseArray()
        pos += 1  -- skip [
        local arr = {}
        skipWS()
        if s:sub(pos,pos) == ']' then pos += 1; return arr end
        while true do
            skipWS()
            arr[#arr+1] = parseValue()
            skipWS()
            local c = s:sub(pos, pos)
            if c == ']' then pos += 1; break
            elseif c == ',' then pos += 1
            else break end
        end
        return arr
    end

    local function parseObject()
        pos += 1  -- skip {
        local obj = {}
        skipWS()
        if s:sub(pos,pos) == '}' then pos += 1; return obj end
        while true do
            skipWS()
            if s:sub(pos,pos) ~= '"' then break end
            local key = parseString()
            skipWS()
            if s:sub(pos,pos) == ':' then pos += 1 end
            skipWS()
            local val = parseValue()
            obj[key] = val
            skipWS()
            local c = s:sub(pos, pos)
            if c == '}' then pos += 1; break
            elseif c == ',' then pos += 1
            else break end
        end
        return obj
    end

    parseValue = function()
        skipWS()
        local c = s:sub(pos, pos)
        if c == '"' then
            return parseString()
        elseif c == '{' then
            return parseObject()
        elseif c == '[' then
            return parseArray()
        elseif c == 't' then
            pos += 4; return true
        elseif c == 'f' then
            pos += 5; return false
        elseif c == 'n' then
            pos += 4; return nil
        elseif c == '-' or c:match("%d") then
            return parseNumber()
        end
        return nil
    end

    local ok, result = pcall(parseValue)
    if ok then return result end
    return nil
end

local function cfgToTable()
    local t = {}
    for _, k in ipairs(SAVE_KEYS) do
        local v = cfg[k]
        if v == nil then
            -- skip nil values
        elseif type(v) == "boolean" or type(v) == "number" or type(v) == "string" then
            t[k] = v
        elseif typeof(v) == "EnumItem" then
            t[k] = {__type = "KeyCode", name = v.Name}
        elseif typeof(v) == "Color3" then
            t[k] = {
                __type = "Color3",
                r = math.floor(v.R * 255 + 0.5),
                g = math.floor(v.G * 255 + 0.5),
                b = math.floor(v.B * 255 + 0.5),
            }
        end
    end
    return t
end

local function tableToCode(loaded)
    local out = {}
    for k, v in pairs(loaded) do
        if type(v) == "table" and v.__type == "KeyCode" then
            local ok2, kc = pcall(function() return Enum.KeyCode[v.name or "Unknown"] end)
            out[k] = (ok2 and kc) and kc or Enum.KeyCode.Unknown
        elseif type(v) == "table" and v.__type == "Color3" then
            out[k] = Color3.fromRGB(v.r or 0, v.g or 0, v.b or 0)
        else
            out[k] = v
        end
    end
    return out
end

local function saveConfig(name)
    local path = CONFIG_FOLDER .. "/" .. name .. CONFIG_EXT
    local ok, err = pcall(function()
        fsWrite(path, jsonEncode(cfgToTable()))
    end)
    return ok, err
end

local function loadConfig(name)
    local path = CONFIG_FOLDER .. "/" .. name .. CONFIG_EXT
    if not pcall(function() return fsExists(path) end) then
        return false, "file not found"
    end
    local ok, err = pcall(function()
        local data = fsRead(path)
        if not data or data == "" then error("empty file") end
        local decoded = jsonDecode(data)
        if type(decoded) ~= "table" then error("parse failed") end
        local converted = tableToCode(decoded)
        for k, v in pairs(converted) do
            cfg[k] = v
        end
    end)
    return ok, err
end

local function deleteConfig(name)
    local path = CONFIG_FOLDER .. "/" .. name .. CONFIG_EXT
    local ok = pcall(function()
        if syn and syn.delete_file then syn.delete_file(path)
        elseif delfile then delfile(path)
        elseif deletefile then deletefile(path) end
    end)
    return ok
end

local function listConfigs()
    local names = {}
    local ok, files = pcall(fsList, CONFIG_FOLDER)
    if not ok then return names end
    for _, f in ipairs(files or {}) do
        local name = tostring(f)
        name = name:match("[/\\]?([^/\\]+)$") or name
        if name:sub(-#CONFIG_EXT) == CONFIG_EXT then
            table.insert(names, name:sub(1, -#CONFIG_EXT - 1))
        end
    end
    table.sort(names)
    return names
end

local cam = game:GetService("Workspace").CurrentCamera
local RunService = game:GetService("RunService")
 
local function wallBetween(targetPos)
    local rayOrigin = cam.CFrame.Position
    local rayDir = (targetPos - rayOrigin)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {lp.Character}
    params.FilterType = Enum.RaycastFilterType.Exclude
    local result = game:GetService("Workspace"):Raycast(rayOrigin, rayDir, params)
    if result and not result.Instance:IsDescendantOf(game:GetService("Workspace"):FindFirstChild(
        game:GetService("Players"):GetPlayerFromCharacter(result.Instance.Parent) and result.Instance.Parent.Name or "")) then
        return true
    end
    return false
end
 
local function getPredicted(part)
    if not part then return nil end
    local vel = part.AssemblyLinearVelocity
    local predX = vel.X * cfg.camlockPredX
    local predY = vel.Y * cfg.camlockPredY
    if cfg.camlockJumpPredEnabled and vel.Y > 0 then predY = predY + vel.Y * cfg.camlockJumpPredY end
    if cfg.camlockFallPredEnabled and vel.Y < 0 then predY = predY + vel.Y * cfg.camlockFallPredY end
    return part.Position + Vector3.new(predX, predY, 0)
end
 
local function getTargets(teamCheck)
    local targets = {}

    -- real players
    for _, player in ipairs(game:GetService("Players"):GetPlayers()) do
        if player == lp then continue end
        if teamCheck and player.Team == lp.Team then continue end
        local char = player.Character
        if char and char:FindFirstChildOfClass("Humanoid") and char:FindFirstChild("HumanoidRootPart") then
            table.insert(targets, { char = char, isBot = false })
        end
    end

    -- NPCs / bots (workspace humanoids not belonging to any player)
    if cfg.camlockTargetBots then
        local playerChars = {}
        for _, p in ipairs(game:GetService("Players"):GetPlayers()) do
            if p.Character then playerChars[p.Character] = true end
        end

        for _, obj in ipairs(game:GetService("Workspace"):GetDescendants()) do
            if obj:IsA("Humanoid") and obj.Health > 0 then
                local char = obj.Parent
                if char and not playerChars[char] then
                    local root = char:FindFirstChild("HumanoidRootPart")
                    if root then
                        table.insert(targets, { char = char, isBot = true })
                    end
                end
            end
        end
    end

    return targets
end

local function closestToCursor(teamCheck, maxDist)
    local closestChar, closestDist = nil, math.huge
    local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)

    for _, entry in ipairs(getTargets(teamCheck)) do
        local root = entry.char:FindFirstChild("HumanoidRootPart")
        if not root then continue end
        if (root.Position - cam.CFrame.Position).Magnitude > maxDist then continue end
        local screenPos, onScreen = cam:WorldToScreenPoint(root.Position)
        if not onScreen then continue end
        local dist = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
        if dist < closestDist then
            closestDist = dist
            closestChar = entry.char
        end
    end

    -- return a fake "player-like" table the camlock logic can use
    if closestChar then
        return { Character = closestChar }
    end
    return nil
end
 
pcall(function() RunService:UnbindFromRenderStep("CamlockStep") end)
RunService:BindToRenderStep("CamlockStep", Enum.RenderPriority.Camera.Value+1, function()
    if cfg.camlock and cfg.camlockMode == "Always" and not camlockTarget then
        camlockTarget = closestToCursor(cfg.camlockTeamCheck, cfg.camlockMaxDist)
    end
    if not cfg.camlock or not camlockTarget then
        if cam.CameraType == Enum.CameraType.Scriptable then cam.CameraType = Enum.CameraType.Custom end
        return
    end
    local char = camlockTarget.Character
    if not char then camlockTarget=nil; cam.CameraType=Enum.CameraType.Custom; return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then camlockTarget=nil; cam.CameraType=Enum.CameraType.Custom; return end
    if cfg.camlockKOCheck then
        local bodyEffects = char:FindFirstChild("BodyEffects")
        if bodyEffects then
            local ko = bodyEffects:FindFirstChild("K.O")
            if ko and ko.Value == true then camlockTarget=nil; cam.CameraType=Enum.CameraType.Custom; return end
        end
        if hum.Health <= 0 then camlockTarget=nil; cam.CameraType=Enum.CameraType.Custom; return end
    end
    local partName = cfg.camlockPart
    if cfg.camlockUseAirPart and hum.FloorMaterial == Enum.Material.Air then partName = cfg.camlockAirPart end
    local part
    if cfg.camlockClosestPart then
        local closestDist2 = math.huge
        for _, p in pairs(char:GetChildren()) do
            if p:IsA("BasePart") then
                local d = (p.Position - cam.CFrame.Position).Magnitude
                if d < closestDist2 then closestDist2=d; part=p end
            end
        end
    else
        part = char:FindFirstChild(partName) or char:FindFirstChild("HumanoidRootPart")
    end
    if not part then return end
    if cfg.camlockWallCheck and wallBetween(part.Position) then return end
    local pred = getPredicted(part)
    if not pred then return end
    local from = cam.CFrame
    local to = CFrame.new(from.Position, pred)
    local t = cfg.camlockSmooth
    local style = cfg.camlockAimStyle
    local easedT = t
    if style=="Linear" then easedT=t
    elseif style=="Sine In" then easedT=1-math.cos((t*math.pi)/2)
    elseif style=="Sine Out" then easedT=math.sin((t*math.pi)/2)
    elseif style=="Sine In-Out" then easedT=-(math.cos(math.pi*t)-1)/2
    elseif style=="Quad In" then easedT=t*t
    elseif style=="Quad Out" then easedT=1-(1-t)*(1-t)
    elseif style=="Quad In-Out" then easedT=t<0.5 and 2*t*t or 1-(-2*t+2)^2/2
    elseif style=="Cubic In" then easedT=t*t*t
    elseif style=="Cubic Out" then easedT=1-(1-t)^3
    elseif style=="Cubic In-Out" then easedT=t<0.5 and 4*t*t*t or 1-(-2*t+2)^3/2
    elseif style=="Quart In" then easedT=t*t*t*t
    elseif style=="Quart Out" then easedT=1-(1-t)^4
    elseif style=="Quart In-Out" then easedT=t<0.5 and 8*t*t*t*t or 1-(-2*t+2)^4/2
    elseif style=="Quint In" then easedT=t*t*t*t*t
    elseif style=="Quint Out" then easedT=1-(1-t)^5
    elseif style=="Quint In-Out" then easedT=t<0.5 and 16*t*t*t*t*t or 1-(-2*t+2)^5/2
    elseif style=="Expo In" then easedT=t==0 and 0 or 2^(10*t-10)
    elseif style=="Expo Out" then easedT=t==1 and 1 or 1-2^(-10*t)
    elseif style=="Expo In-Out" then easedT=t==0 and 0 or t==1 and 1 or t<0.5 and 2^(20*t-10)/2 or (2-2^(-20*t+10))/2
    elseif style=="Circ In" then easedT=1-math.sqrt(1-t^2)
    elseif style=="Circ Out" then easedT=math.sqrt(1-(t-1)^2)
    elseif style=="Circ In-Out" then easedT=t<0.5 and (1-math.sqrt(1-(2*t)^2))/2 or (math.sqrt(1-(-2*t+2)^2)+1)/2
    elseif style=="Back In" then local c1=1.70158;local c3=c1+1;easedT=c3*t*t*t-c1*t*t
    elseif style=="Back Out" then local c1=1.70158;local c3=c1+1;easedT=1+c3*(t-1)^3+c1*(t-1)^2
    elseif style=="Back In-Out" then local c1=1.70158;local c2=c1*1.525;easedT=t<0.5 and((2*t)^2*((c2+1)*2*t-c2))/2 or((2*t-2)^2*((c2+1)*(t*2-2)+c2)+2)/2
    elseif style=="Elastic In" then local c4=(2*math.pi)/3;easedT=t==0 and 0 or t==1 and 1 or -2^(10*t-10)*math.sin((t*10-10.75)*c4)
    elseif style=="Elastic Out" then local c4=(2*math.pi)/3;easedT=t==0 and 0 or t==1 and 1 or 2^(-10*t)*math.sin((t*10-0.75)*c4)+1
    elseif style=="Elastic In-Out" then local c5=(2*math.pi)/4.5;easedT=t==0 and 0 or t==1 and 1 or t<0.5 and -(2^(20*t-10)*math.sin((20*t-11.125)*c5))/2 or(2^(-20*t+10)*math.sin((20*t-11.125)*c5))/2+1
    elseif style=="Bounce In" then
        local function bo(x) local n1,d1=7.5625,2.75;if x<1/d1 then return n1*x*x elseif x<2/d1 then x=x-1.5/d1;return n1*x*x+0.75 elseif x<2.5/d1 then x=x-2.25/d1;return n1*x*x+0.9375 else x=x-2.625/d1;return n1*x*x+0.984375 end end
        easedT=1-bo(1-t)
    elseif style=="Bounce Out" then
        local n1,d1=7.5625,2.75;if t<1/d1 then easedT=n1*t*t elseif t<2/d1 then t=t-1.5/d1;easedT=n1*t*t+0.75 elseif t<2.5/d1 then t=t-2.25/d1;easedT=n1*t*t+0.9375 else t=t-2.625/d1;easedT=n1*t*t+0.984375 end
    elseif style=="Bounce In-Out" then
        local function bo(x) local n1,d1=7.5625,2.75;if x<1/d1 then return n1*x*x elseif x<2/d1 then x=x-1.5/d1;return n1*x*x+0.75 elseif x<2.5/d1 then x=x-2.25/d1;return n1*x*x+0.9375 else x=x-2.625/d1;return n1*x*x+0.984375 end end
        easedT=t<0.5 and(1-bo(1-2*t))/2 or(1+bo(2*t-1))/2
    elseif style=="Spring" then easedT=math.clamp(math.sin(t*math.pi)*0.3+t,0,1)
    elseif style=="Velocity" then
        local angle=math.acos(math.clamp(from.LookVector:Dot(to.LookVector),-1,1))
        easedT=math.clamp(t+angle*0.5,0.01,1)
    end
    cam.CFrame = from:Lerp(to, easedT)
end)
 
UIS.InputBegan:Connect(function(i, gp)
    if gp then return end
    if i.KeyCode == cfg.camlockKey then
        if cfg.camlockMode == "Toggle" then
            cfg.camlock = not cfg.camlock
            if not cfg.camlock then camlockTarget=nil; cam.CameraType=Enum.CameraType.Custom
            else camlockTarget = closestToCursor(cfg.camlockTeamCheck, cfg.camlockMaxDist) end
        elseif cfg.camlockMode == "Hold" then
            cfg.camlock = true
            camlockTarget = closestToCursor(cfg.camlockTeamCheck, cfg.camlockMaxDist)
        end
    end
    if i.KeyCode == cfg.walkspeedKey then
        cfg.walkspeedException = not cfg.walkspeedException
        if cfg.walkspeedException then applyWalkspeed() else disableWalkspeed() end
    end
end)
 
UIS.InputEnded:Connect(function(i, gp)
    if gp then return end
    if i.KeyCode == cfg.camlockKey and cfg.camlockMode == "Hold" then
        cfg.camlock = false; camlockTarget = nil; cam.CameraType = Enum.CameraType.Custom
    end
end)
